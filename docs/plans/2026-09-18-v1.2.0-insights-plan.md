### A1. 数据结构（全部 `#[derive(Debug, Serialize)]`，snake_case 直达前端）

```rust
#[derive(Debug, Serialize)]
pub struct HourCell {
    pub date: String,     // "2026-09-15"
    pub hour: i64,        // 0–23
    pub events: i64,      // 键鼠总事件，口径同周账单 act_events
    pub front_secs: i64,  // 该小时前台应用秒（app_usage_hourly 求和；监控关=0）
    pub audio_secs: i64,  // 该小时音频秒（audio_usage_hourly 求和；监控关=0）
}

#[derive(Debug, Serialize)]
pub struct HourHeatmap {
    pub week_start: String,
    pub week_end: String,
    pub cells: Vec<HourCell>,   // 只含有数据的小时格
    pub peak_hour: Option<i64>, // 跨天按小时聚合 events 最大者；无数据 None
}

#[derive(Debug, Serialize)]
pub struct TrendPoint {
    pub week_start: String,
    pub income: f64,             // WeekBill.total_income（口径与账单页一致）
    pub slack_rate: Option<f64>, // Σslack_seconds ÷ front_seconds；front=0 → None
}

#[derive(Debug, Serialize)]
pub struct WeekTrend {
    pub week_start: String,       // 窗口最早周（offset+7）
    pub points: Vec<TrendPoint>,  // 旧→新，8 个
}

#[derive(Debug, Serialize)]
pub struct BodyDay {
    pub date: String,
    pub weekday: String, // "周一"
    pub events: i64,
}

#[derive(Debug, Serialize)]
pub struct BodyBill {
    pub week_start: String,
    pub week_end: String,
    pub clicks: i64,      // left − dbl + right + mid + xbtn（双击折算 1 次，同周账单）
    pub keys: i64,
    pub moves: i64,
    pub pixels: i64,      // 前端按 96dpi 换算米
    pub wheel_ticks: i64,
    pub days: Vec<BodyDay>,       // 恒 7 个（无记录天 events=0）
    pub busiest: Option<BodyDay>, // events 最大且 >0
}
```

### A3. SQL（字段名以 db.rs DDL 为准：act_hourly = moves/pixels/`left`/dbl/`right`/wheel/wheel_ticks/mid/xbtn/keys）

```sql
-- 热力图①键鼠：按 (date,hour)
SELECT date, hour,
       SUM(moves)+SUM(`left`)+SUM(dbl)+SUM(`right`)+SUM(wheel)+SUM(mid)+SUM(xbtn)+SUM(keys) AS events
  FROM act_hourly WHERE date BETWEEN ?1 AND ?2 GROUP BY date, hour;

-- 热力图②前台秒
SELECT date, hour, SUM(seconds) FROM app_usage_hourly WHERE date BETWEEN ?1 AND ?2 GROUP BY date, hour;

-- 热力图③音频秒
SELECT date, hour, SUM(seconds) FROM audio_usage_hourly WHERE date BETWEEN ?1 AND ?2 GROUP BY date, hour;
-- 三结果 Rust 侧按 (date,hour) 合并进 Vec<HourCell>（BTreeMap<(String,i64), HourCell>）

-- 身体账单①按天
SELECT date,
       SUM(moves)+SUM(`left`)+SUM(dbl)+SUM(`right`)+SUM(wheel)+SUM(mid)+SUM(xbtn)+SUM(keys) AS events
  FROM act_hourly WHERE date BETWEEN ?1 AND ?2 GROUP BY date;
-- 身体账单②全区间五指标
SELECT COALESCE(SUM(`left`)-SUM(dbl)+SUM(`right`)+SUM(mid)+SUM(xbtn),0) AS clicks,
       COALESCE(SUM(keys),0) AS keys,
       COALESCE(SUM(moves),0) AS moves,
       COALESCE(SUM(pixels),0) AS pixels,
       COALESCE(SUM(wheel_ticks),0) AS wheel_ticks
  FROM act_hourly WHERE date BETWEEN ?1 AND ?2;
```

- `weekday` 中文标签：本地 `const WEEKDAYS: [&str; 7]`（周一..周日）+ `weekday().num_days_from_monday()` 索引（与 weekbill 同法）
- 恒 7 个 BodyDay：从 ws 迭代 7 天 LEFT JOIN 聚合结果（BTreeMap 按 date 查）

## 口径速查（实现中随时对照）

| 口径 | 规则 |
|---|---|
| 周区间 | ws = week_start_of(today − 7×offset)，we = ws+6；offset<0 封顶 0 |
| events（热力/身体 events 列） | moves+left+dbl+right+wheel+mid+xbtn+keys（同周账单 act_events） |
| clicks | left − dbl + right + mid + xbtn（双击折算 1 次，同周账单） |
| 趋势窗口 | [offset, offset+7] 共 8 周，旧→新；income 复用 weekbill::week_bill |
| slack_rate | Σdays.slack_seconds ÷ front_seconds；front=0 → None（断线不画 0） |
| 无记录周 income | 0 照画（与账单口径一致） |
| 热力强度 | 前端 sqrt(v/max) → 1+min(3, floor(r×4))，v≤0 → 0 级 |
| 像素→米 | px × 2.54 ÷ 96 ÷ 100（96dpi 估算，UI 注明） |
| 远程排除 | 后端字段已有（默认 true）；前端 readCfg/load/change→saveNow 三处接线 |
| 懒加载 | 切 tab / 翻周才拉；不接 tick 轮询 |
